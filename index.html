    Browser API

    The same canonical engine backs the UI and window.faultline. Inspect first, then load a complete case atomically or edit one axis with an expected revision.

    const state = await window.faultline.inspect()

await window.faultline.loadCase({
  expectedRevision: state.revision,
  case: {
    html: '<main>...</main>',
    css: 'main { ... }',
    js: '...',
    oracle: {
      kind: 'dom_exists',
      selector: 'main',
      equals: true,
      action: { kind: 'none' }
    }
  }
})

await window.faultline.applySource({
  expectedRevision,
  targetAxis,
  source
})
